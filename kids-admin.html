First-time setup (do this once, as the very first admin) — and three rules updates since

Already set up? The rules below now add: (1) a third role, teacher, used by the Tuhfatul Atfaal Teacher Portal — regular admins can approve teachers, and teacher accounts can never delete a student; and (2) a kids_quest_usernames collection that lets a student's own email (if they gave one at sign-up) become their real login, so "Forgot your password?" can email them directly instead of only the admin; and (3) a kids_quest_quizzes collection for quizzes teachers write in the Teacher Portal. Paste the whole updated rules block below into the Firebase console even if you already did the original setup; steps 2–3 only apply the very first time.

  1. Add the Firestore security rules. In Firebase Console → Firestore Database → Rules, add these two blocks inside your existing match /databases/{database}/documents { ... } (alongside whatever rules already exist for other collections — don't remove those): 
    match /kids_quest_admins/{adminEmail} {
  // anyone signed in can check their OWN status (admin, super or teacher)
  allow get: if request.auth != null
    && request.auth.token.email.lower() == adminEmail;
  // a super admin can see and manage the whole list — admins, super
  // admins and teachers alike
  allow get, list, create, update, delete: if request.auth != null
    && get(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower())).data.role == 'super';
  // a regular admin can read the whole list too (so it can tell teacher
  // rows apart from admin/super rows), and can approve, add or remove
  // TEACHER accounts specifically — but can never touch an admin/super doc
  allow get, list: if request.auth != null
    && get(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower())).data.role == 'admin';
  allow create, update: if request.auth != null
    && request.resource.data.role == 'teacher'
    && get(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower())).data.role == 'admin';
  allow delete: if request.auth != null
    && resource.data.role == 'teacher'
    && get(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower())).data.role == 'admin';
  // anyone on the list can edit ONLY their own full name and contact
  // email (shown on certificates they sign) — never their role
  allow update: if request.auth != null
    && request.auth.token.email.lower() == adminEmail
    && request.resource.data.role == resource.data.role
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['fullName', 'contactEmail', 'profileUpdatedAt']);
}
match /kids_quest_students/{uid} {
  // a student can only ever read/write their own saved progress
  allow get, create, update: if request.auth != null && request.auth.uid == uid;
  // admins and super admins can view and edit every student
  allow get, list, create, update: if isAdminOrSuper();
  // a teacher only sees students linked to them — students who joined
  // one of their classes, or accounts that teacher created
  allow get, list, update: if isTeacher()
    && request.auth.token.email.lower() in resource.data.get('teacherEmails', []);
  allow create: if isTeacher()
    && request.auth.token.email.lower() in request.resource.data.get('teacherEmails', []);
  // only an admin or super admin can delete a student outright
  allow delete: if isAdminOrSuper();
}
function staffRole() {
  return get(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower())).data.get('role', 'admin');
}
function isAdminOrSuper() {
  return isStaff() && staffRole() in ['admin', 'super'];
}
function isTeacher() {
  return isStaff() && staffRole() == 'teacher';
}
match /kids_quest_usernames/{username} {
  // public read — needed to resolve a username to its login email
  // BEFORE signing in, since a student who gave a real email at sign-up
  // no longer uses the deterministic synthetic address. The value here
  // is just an email address, no more sensitive than the synthetic
  // addresses it replaces.
  allow get: if true;
  // the account itself, right after creating its own login, can claim
  // its own username (pointing only at its OWN email) — or an admin,
  // super admin or teacher can create one on its behalf (needed when
  // renaming a student who already graduated to a real email)
  allow create: if request.auth != null
    && (request.resource.data.email == request.auth.token.email.lower()
        || exists(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower())));
  // any listed admin, super admin or teacher can update/remove a
  // mapping (needed when renaming a student's username)
  allow update, delete: if request.auth != null
    && exists(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower()));
}
match /kids_quest_quizzes/{quizId} {
  // any signed-in student can read quizzes, to take them
  allow get, list: if request.auth != null;
  // only a listed admin, super admin or teacher can write them
  allow create, update, delete: if request.auth != null
    && exists(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower()));
}
function isStaff() {
  return request.auth != null
    && exists(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower()));
}
// teacher announcements, each with a class comment section
match /kids_quest_announcements/{annId} {
  allow get, list: if request.auth != null;
  allow create, update, delete: if isStaff();
  match /comments/{commentId} {
    allow get, list: if request.auth != null;
    // your own uid only; only staff may post as 'teacher'; students only
    // while the teacher has comments open
    allow create: if request.auth != null
      && request.resource.data.uid == request.auth.uid
      && request.resource.data.text is string
      && request.resource.data.text.size() > 0
      && request.resource.data.text.size() <= 1000
      && (request.resource.data.role == 'student'
          || (request.resource.data.role == 'teacher' && isStaff()))
      && (isStaff() || get(/databases/$(database)/documents/kids_quest_announcements/$(annId)).data.commentsOpen == true);
    allow delete: if request.auth != null
      && (resource.data.uid == request.auth.uid || isStaff());
  }
}
// private teacher <-> student messages: only that student and staff
// staff who may see one student's private things: admins and super
// admins, or a teacher linked to that student (joined their class /
// created by them)
function canStaffStudent(studentUid) {
  return isAdminOrSuper() || (isTeacher()
    && request.auth.token.email.lower() in get(/databases/$(database)/documents/kids_quest_students/$(studentUid)).data.get('teacherEmails', []));
}
// private teacher <-> student messages: only that student, admins, and
// the student's own teachers
match /kids_quest_messages/{studentUid} {
  allow get, create, update: if request.auth != null
    && (request.auth.uid == studentUid || canStaffStudent(studentUid));
  allow list: if isAdminOrSuper();
  allow delete: if canStaffStudent(studentUid);
  match /thread/{messageId} {
    allow get, list: if request.auth != null
      && (request.auth.uid == studentUid || canStaffStudent(studentUid));
    allow create: if request.auth != null
      && request.resource.data.text is string
      && request.resource.data.text.size() > 0
      && request.resource.data.text.size() <= 2000
      && ((request.auth.uid == studentUid && request.resource.data.from == 'student')
          || (canStaffStudent(studentUid) && request.resource.data.from == 'teacher'));
    allow delete: if canStaffStudent(studentUid);
  }
}
match /kids_quest_classes/{classId} {
  // anyone signed in can open a class by its code (to join it)
  allow get: if request.auth != null;
  // staff see every class; a student can list only classes they're in
  allow list: if isStaff()
    || (request.auth != null && request.auth.uid in resource.data.studentUids);
  // staff create classes in their own name
  allow create: if isStaff()
    && request.resource.data.teacherEmail == request.auth.token.email.lower();
  // the class's own teacher, or an admin / super admin, manages it
  allow update, delete: if isStaff()
    && (resource.data.teacherEmail == request.auth.token.email.lower()
        || get(/databases/$(database)/documents/kids_quest_admins/$(request.auth.token.email.lower())).data.role in ['admin', 'super']);
  // a student may only add or remove THEMSELVES (join / leave)
  allow update: if request.auth != null
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['studentUids'])
    && request.resource.data.studentUids.toSet().difference(resource.data.studentUids.toSet()).hasOnly([request.auth.uid])
    && resource.data.studentUids.toSet().difference(request.resource.data.studentUids.toSet()).hasOnly([request.auth.uid]);
}
// homework with due dates: students read, staff write
match /kids_quest_homework/{hwId} {
  allow get, list: if request.auth != null;
  allow create, update, delete: if isStaff();
}
// private parent pages: anyone with the link can read ONE page by its
// secret id; nobody can list them. The student writes their own `data`;
// only staff write `staffData` (attendance).
match /kids_quest_parent/{token} {
  allow get: if true;
  allow create: if request.auth != null
    && (isStaff() || (request.resource.data.uid == request.auth.uid
        && !('staffData' in request.resource.data)));
  allow update: if request.auth != null
    && (isStaff() || (resource.data.uid == request.auth.uid
        && request.resource.data.uid == request.auth.uid
        && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['staffData'])));
  allow delete: if isStaff();
}
// Talqeen voice recordings: only that student and staff
match /kids_quest_recordings/{recId} {
  allow get, list: if isStaff()
    || (request.auth != null && resource.data.studentUid == request.auth.uid);
  allow create: if request.auth != null
    && request.resource.data.data is string
    && request.resource.data.data.size() < 1000000
    && ((isStaff() && request.resource.data.from == 'teacher')
        || (request.resource.data.from == 'student' && request.resource.data.studentUid == request.auth.uid));
  allow update: if (isStaff() || (request.auth != null && resource.data.studentUid == request.auth.uid))
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['heard', 'heardAt', 'feedback', 'feedbackBy', 'feedbackAt']);
  allow delete: if isStaff()
    || (request.auth != null && resource.data.studentUid == request.auth.uid && resource.data.from == 'student');
}
// class attendance: staff only
match /kids_quest_attendance/{sessionId} {
  allow get, list, create, update, delete: if isStaff();
}
// class leaderboards: members read; each member writes only their own row
match /kids_quest_boards/{classId} {
  function isMember() {
    return request.auth != null
      && request.auth.uid in get(/databases/$(database)/documents/kids_quest_classes/$(classId)).data.studentUids;
  }
  allow get: if isStaff() || isMember();
  allow create: if isMember()
    && request.resource.data.keys().hasOnly(['entries'])
    && request.resource.data.entries.keys().hasOnly([request.auth.uid]);
  allow update: if isStaff() || (isMember()
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['entries'])
    && request.resource.data.entries.diff(resource.data.entries).affectedKeys().hasOnly([request.auth.uid]));
  allow delete: if isStaff();
}
